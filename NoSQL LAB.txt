db.createCollection("customers", {

        $jsonSchema: {

            properties: {

                name: { bsonType: "string" },
                age: { bsonType: "int", minimum: 18 },
                phone: { bsonType: "string" }
            }
        }
    }
})

db.createCollection("medicines", {
    validator: {

            bsonType: "object",
            required: ["medId", "medName", "supplierId", "price", "stock", "expiryDate"],
            properties: {
                medId: { bsonType: "int" },
                medName: { bsonType: "string" },
                supplierId: { bsonType: "int" },
                price: { bsonType: "int" },

// i. List all orders with customer and medicine names
db.orders.aggregate([
  {$lookup:{from:"customers",localField:"custId",foreignField:"custId",as:"c"}},
  {$unwind:"$c"},
  {$lookup:{from:"medicines",localField:"medId",foreignField:"medId",as:"m"}},
  {$unwind:"$m"},
  {$project:{_id:0,orderId:1,customer:"$c.name",medicine:"$m.medName"}}
])

// ii. Total sales per medicine
db.orders.aggregate([
  {$lookup:{from:"medicines",localField:"medId",foreignField:"medId",as:"m"}},
  {$unwind:"$m"},
  {$group:{
    _id:"$m.medName",
    sold:{$sum:"$quantity"},
    revenue:{$sum:{$multiply:["$quantity","$m.price"]}}
  }}
])

// iii. Supplier-wise revenue
db.orders.aggregate([
  {$lookup:{from:"medicines",localField:"medId",foreignField:"medId",as:"m"}},
  {$unwind:"$m"},
  {$lookup:{from:"suppliers",localField:"m.supplierId",foreignField:"supplierId",as:"s"}},
  {$unwind:"$s"},
  {$group:{
    _id:"$s.name",
    revenue:{$sum:{$multiply:["$quantity","$m.price"]}}
  }}
])

// iv. Medicines below stock threshold (less than 70)
db.medicines.aggregate([
  {$match:{stock:{$lt:70}}},
  {$lookup:{from:"suppliers",localField:"supplierId",foreignField:"supplierId",as:"s"}},
  {$unwind:"$s"},
  {$project:{_id:0,medName:1,stock:1,supplier:"$s.name"}}
])

// v. Customer order summary
db.customers.aggregate([
  {$lookup:{from:"orders",localField:"custId",foreignField:"custId",as:"o"}},
  {$project:{_id:0,name:1,totalOrders:{$size:"$o"}}},
  {$sort:{totalOrders:-1}}
])

import tkinter as tk
from pymongo import MongoClient
col = MongoClient("mongodb://localhost:27017/").collegeDB.students

def insert():
    col.insert_one({"name": e[0].get(), "roll": e[1].get(),
                    "branch": e[2].get(), "cgpa": float(e[3].get())})
    for x in e: x.delete(0, tk.END)

def show():
    t.delete("1.0", tk.END)
    for s in col.find():
        t.insert(tk.END, f"{s['name']} | {s['roll']} | {s['branch']} | {s['cgpa']}\n")
